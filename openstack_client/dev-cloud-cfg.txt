package_update: true
package_upgrade: true

packages:
  - git
  - python3
  - python3-pip
  - curl
  - unzip
  - openjdk-11-jre-headless
  - software-properties-common
  - openjdk-17-jdk

runcmd:
  - echo "Installing Apache Pulsar..."
  - mkdir -p /opt/pulsar
  - wget https://archive.apache.org/dist/pulsar/pulsar-2.9.4/apache-pulsar-2.9.4-bin.tar.gz -O /opt/pulsar/pulsar.tar.gz
  - tar -xvf /opt/pulsar/pulsar.tar.gz -C /opt/pulsar
  - mv /opt/pulsar/apache-pulsar-2.9.4 /opt/pulsar/pulsar

  - echo "Starting Pulsar in standalone mode..."
  - /opt/pulsar/pulsar/bin/pulsar-daemon start standalone

  - echo "Waiting for Pulsar to initialize..."
  - sleep 20

  - echo "Creating tenant and namespace..."
  - JAVA_HOME=/usr/lib/jvm/java-17-openjdk-amd64 /opt/pulsar/pulsar/bin/pulsar-admin tenants create github-tenants -r standalone -cl standalone
  - JAVA_HOME=/usr/lib/jvm/java-17-openjdk-amd64 /opt/pulsar/pulsar/bin/pulsar-admin namespaces create github-tenants/Model-namespace

byobu_default: system
